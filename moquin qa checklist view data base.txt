CREATE VIEW IF NOT EXISTS jobs_readable AS
SELECT
  id,
  json_extract(data,'$.jobNumber')     AS job_number,
  json_extract(data,'$.productNumber') AS product_number,
  json_extract(data,'$.station')       AS station,
  json_extract(data,'$.customer')      AS customer,
  json_extract(data,'$.process')       AS process,
  json_extract(data,'$.status')        AS status,
  json_extract(data,'$.timestamp')     AS started_at,
  updated_at
FROM records
WHERE type = 'job';




After that, any time you want the readable view, just run:

SELECT * FROM jobs_readable;
